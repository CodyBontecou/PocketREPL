### ModelDownloadManager
```swift
actor ModelDownloadManager {
    private var activeDownloads: [String: URLSessionDownloadTask] = [:]
    
    func download(
        model: ModelRegistryEntry,
        onProgress: @escaping (Double) -> Void
    ) async throws -> URL {
        let destination = modelsDirectory.appendingPathComponent("\(model.id).gguf")
        
        // Check if already downloaded
        if FileManager.default.fileExists(atPath: destination.path) {
            return destination
        }
        
        // Download with progress
        let (tempURL, _) = try await URLSession.shared.download(from: model.downloadURL) { progress in
            onProgress(progress.fractionCompleted)
        }
        
        // Move to permanent location
        try FileManager.default.moveItem(at: tempURL, to: destination)
        
        // Verify hash if provided
        if let expectedHash = model.sha256 {
            let actualHash = try computeSHA256(of: destination)
            guard actualHash == expectedHash else {
                try FileManager.default.removeItem(at: destination)
                throw ModelError.loadFailed(reason: "Hash mismatch")
            }
        }
        
        return destination
    }
    
    func cancelDownload(modelId: String) {
        activeDownloads[modelId]?.cancel()
    }
    
    var modelsDirectory: URL {
        FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("Models", isDirectory: true)
    }
}
```
