## Goal
Create a SwiftUI view for browsing, downloading, loading, and managing local models.

## Implementation

### ModelManagementView.swift
```swift
struct ModelManagementView: View {
    @ObservedObject var modelManager: ModelBackendManager
    @StateObject private var downloadManager = ModelDownloadManager()
    
    var body: some View {
        List {
            // Current Model Section
            Section("Active Model") {
                if let info = modelManager.modelInfo {
                    ActiveModelRow(info: info, state: modelManager.state)
                    Button("Unload") { Task { await modelManager.unload() } }
                } else {
                    Text("No model loaded")
                        .foregroundStyle(.secondary)
                }
            }
            
            // Installed Models Section
            Section("Installed") {
                ForEach(installedModels) { model in
                    InstalledModelRow(model: model) {
                        Task { await loadModel(model) }
                    }
                }
                .onDelete(perform: deleteModels)
            }
            
            // Available Downloads Section
            Section("Available") {
                ForEach(availableModels) { model in
                    DownloadableModelRow(model: model) {
                        Task { await downloadModel(model) }
                    }
                }
            }
        }
        .navigationTitle("Models")
    }
}
```

### Loading Progress View
```swift
struct ModelLoadingOverlay: View {
    let state: ModelState
    
    var body: some View {
        switch state {
        case .loading(let progress):
            VStack(spacing: 16) {
                ProgressView(value: progress)
                    .progressViewStyle(.linear)
                Text("Loading model... \(Int(progress * 100))%")
            }
            .padding()
            .background(.regularMaterial)
            .cornerRadius(12)
            
        case .generating:
            HStack {
                ProgressView()
                Text("Generating...")
            }
            
        default:
            EmptyView()
        }
    }
}
```

### Download Progress Row
```swift
struct DownloadProgressRow: View {
    let model: ModelRegistryEntry
    let progress: Double
    let onCancel: () -> Void
    
    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            HStack {
                Text(model.name)
                Spacer()
                Button("Cancel", action: onCancel)
                    .buttonStyle(.borderless)
            }
            ProgressView(value: progress)
            Text("\(Int(progress * 100))% of \(formatBytes(model.sizeBytes))")
                .font(.caption)
                .foregroundStyle(.secondary)
        }
    }
}
```
